\subsection{双谱分离系统设计}

\begin{frame}{双谱分离原理}
\frametitle{核心技术：双谱分离系统}
\begin{columns}
\column{0.5\textwidth}
\begin{block}{物理原理}
天空辐射包含两个主要成分：
\begin{itemize}
    \item \textbf{太阳短波辐射（SW）}：0.3-3 $\mu$m，主要是可见光和近红外
    \item \textbf{大气长波辐射（LW）}：3-100 $\mu$m，主要是热红外辐射
\end{itemize}
\end{block}

\begin{block}{分离方法}
通过\textbf{双通道测量} + \textbf{差分计算}：
\begin{align*}
V_A &\approx k \cdot (P_{SW} + P_{LW}) \\
V_B &\approx k \cdot P_{LW} \\
V_{SW} &= V_A - V_B
\end{align*}
\end{block}

\column{0.5\textwidth}
\begin{block}{实现方案}
\textbf{装置A（总辐射通道）}：
\begin{itemize}
    \item 涂黑热电芯片（吸收全波段）
    \item 无滤光片
\end{itemize}

\textbf{装置B（长波通道）}：
\begin{itemize}
    \item 涂黑热电芯片
    \item 加装长波窗（如黑色聚乙烯薄膜）
    \item 阻挡短波，透过长波
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{聚光装置设计}
\frametitle{聚光装置：六边形平面镜阵列}
\begin{columns}
\column{0.6\textwidth}
\begin{block}{设计特点}
\begin{itemize}
    \item 采用2024秋季组改进的六边形平面镜阵列
    \item 多镜面聚光提高辐射强度，增强信号
    \item 两个完全相同的装置分别用于A、B通道
\end{itemize}
\end{block}

\begin{block}{关键参数}
\begin{itemize}
    \item 镜面数量：6-12片
    \item 镜面倾角：根据焦距优化
    \item 焦点直径：匹配热电芯片尺寸
    \item 聚光比：约5-10倍
\end{itemize}
\end{block}

\column{0.4\textwidth}
\begin{block}{材料选择}
\textbf{长波窗材料}：
\begin{itemize}
    \item \textcolor{red}{错误}：玻璃片（阻挡长波）
    \item \textcolor{green}{正确}：黑色聚乙烯薄膜
    \begin{itemize}
        \item 阻挡可见光
        \item 透过中远红外
        \item 成本低廉
    \end{itemize}
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\subsection{冷端稳定与风噪抑制}

\begin{frame}{冷端稳定方案}
\frametitle{解决冷端加热与风噪问题}
\begin{block}{问题分析}
\begin{itemize}
    \item 2024秋季组发现：冷端被日照直接加热，产生系统误差
    \item 2022秋季组担心：风致温度漂移影响测量稳定性
\end{itemize}
\end{block}

\begin{columns}
\column{0.5\textwidth}
\begin{block}{冷端散热方案}
\begin{enumerate}
    \item 涂抹导热硅脂
    \item 紧密贴装大型CPU散热鳍片
    \item 增强冷端散热能力
\end{enumerate}
\end{block}

\begin{block}{遮阳保护}
设计反射罩（硬纸板+铝箔）：
\begin{itemize}
    \item 保护散热鳍片不被太阳直射
    \item 只允许聚光镜反射光束照射热端
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{block}{差分测量抑制风噪}
借鉴2022秋季组思路：
\begin{itemize}
    \item \textbf{芯片1（信号片）}：焦点中心，测量（辐射+风）
    \item \textbf{芯片2（参考片）}：焦点旁，测量（环境+风）
    \item $V_{signal} = V_{芯片1} - V_{芯片2}$
\end{itemize}
差分信号消除共模风噪和环境温度漂移
\end{block}
\end{columns}
\end{frame}

\subsection{自动化数据采集系统}

\begin{frame}{自动化数据采集系统架构}
\frametitle{解决往届手动测量痛点}
\begin{block}{系统组成}
\begin{itemize}
    \item \textbf{主控器}：Raspberry Pi（树莓派）或Arduino/ESP32
    \item \textbf{ADC模块}：ADS1115（16位高精度ADC，内置可编程放大器）
    \item \textbf{传感器}：多路热电芯片
    \item \textbf{数据记录}：自动保存为CSV文件，为CNN训练准备数据集
\end{itemize}
\end{block}

\begin{columns}
\column{0.5\textwidth}
\begin{block}{ADS1115关键特性}
\begin{itemize}
    \item 16位分辨率（vs. Arduino 10位）
    \item 4路差分输入通道
    \item 内置PGA放大器（适合mV级信号）
    \item I2C接口（节省引脚）
    \item 可级联多模块
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{block}{增益设置}
针对热电芯片mV级信号：
\begin{itemize}
    \item GAIN\_SIXTEEN: $\pm$0.256V
    \item 分辨率: 0.0078125 mV/bit
    \item 最适合热电芯片测量
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{数据采集硬件连接}
\frametitle{硬件连接方案}
\begin{columns}
\column{0.5\textwidth}
\begin{block}{树莓派 $\leftrightarrow$ ADS1115}
\begin{tabular}{ll}
\toprule
树莓派引脚 & ADS1115 \\
\midrule
Pin 1 (3.3V) & VDD \\
Pin 6 (GND) & GND \\
Pin 3 (SDA) & SDA \\
Pin 5 (SCL) & SCL \\
\bottomrule
\end{tabular}
\end{block}

\begin{block}{ADS1115 $\leftrightarrow$ 热电芯片}
差分测量模式：
\begin{itemize}
    \item 芯片1：P $\rightarrow$ A0, N $\rightarrow$ A1
    \item 芯片2：P $\rightarrow$ A2, N $\rightarrow$ A3
\end{itemize}
一个ADS1115可同时采集两路差分信号
\end{block}

\column{0.5\textwidth}
\begin{block}{多模块级联}
通过ADDR引脚设置I2C地址：
\begin{itemize}
    \item ADDR $\rightarrow$ GND: 0x48
    \item ADDR $\rightarrow$ VDD: 0x49
    \item ADDR $\rightarrow$ SDA: 0x4A
    \item ADDR $\rightarrow$ SCL: 0x4B
\end{itemize}
最多可级联4个ADS1115，实现8路差分采集
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{数据采集软件实现}
\frametitle{Python数据采集程序}
\begin{block}{推荐方案：树莓派 + Python}
\begin{itemize}
    \item 完整Linux系统，原生支持Python
    \item 可使用PyCharm远程SSH开发
    \item 可直接运行深度学习模型
\end{itemize}
\end{block}

\begin{block}{代码示例（核心部分）}
\begin{lstlisting}[language=Python]
import board, busio
import adafruit_ads1x15.ads1115 as ADS
from adafruit_ads1x15.analog_in import AnalogIn
import csv, time

# 初始化I2C和ADS1115
i2c = busio.I2C(board.SCL, board.SDA)
ads = ADS.ADS1115(i2c)
ads.gain = 16  # GAIN_SIXTEEN

# 差分通道
chan_A = AnalogIn(ads, ADS.P0, ADS.P1)
chan_B = AnalogIn(ads, ADS.P2, ADS.P3)
\end{lstlisting}
\end{block}
\end{frame}

\begin{frame}[fragile]{数据采集与保存}
\frametitle{高频数据采集与CSV保存}
\begin{lstlisting}[language=Python]
# 创建CSV文件并采集数据
with open('radiation_data.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['timestamp', 'V_SW', 'V_LW', 
                     'V_total', 'V_longwave'])
    
    try:
        while True:
            # 读取电压
            v_total = chan_A.voltage * 1000  # 转换为mV
            v_longwave = chan_B.voltage * 1000
            v_sw = v_total - v_longwave  # 短波分量
            
            timestamp = time.time()
            writer.writerow([timestamp, v_sw, v_longwave, 
                           v_total, v_longwave])
            f.flush()  # 立即写入磁盘
            
            time.sleep(0.5)  # 2 Hz采样率
    except KeyboardInterrupt:
        print("采集完成")
\end{lstlisting}
\end{frame}

\begin{block}{项目目标}

\begin{columns}

    \item 同步拍摄天空照片
    \item 自动记录时间戳、图像、$V_{SW}$、$V_{LW}$
    \item 形成大型训练数据集

\end{block}

\begin{alertblock}{超越往届}
\begin{itemize}
    \item 2024组：简单RGB直方图
    \item 2022组：像素点百分比

\column{0.5\textwidth}
\begin{block}{模型架构}
\begin{itemize}
    \item \textbf{输入}：天空图像（RGB）
    \item \textbf{输出}：$[V_{SW}, V_{LW}]$
    \item \textbf{骨干网络}：预训练CNN
    \begin{itemize}
        \item MobileNetV2（轻量）
        \item ResNet50（精确）
    \end{itemize}
    \item \textbf{输出层}：2神经元全连接层
\end{itemize}
\end{block}

\begin{block}{应用价值}
\begin{itemize}
    \item 无需实际测量即可估算辐射
    \item 低成本天气监测方案
    \item 实时云量、辐射分析
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\subsection{时间序列预测}

\begin{frame}{创新项目二：辐射短期预报系统}
\frametitle{基于时间序列的天气变化预测}
\begin{block}{项目目标}
利用高频时间序列数据，预测未来几分钟的辐射变化和天气趋势
\end{block}

\begin{columns}
\column{0.5\textwidth}
\begin{block}{数据特征}
\begin{itemize}
    \item 高频采集：2 Hz（每秒2次）
    \item 双通道：$V_{SW}$、$V_{LW}$
    \item 时间窗口：过去60秒数据
\end{itemize}
\end{block}

\begin{block}{模型方案}
\begin{itemize}
    \item \textbf{输入}：60×2矩阵（过去60秒）
    \item \textbf{输出}：未来15秒$V_{SW}$值
    \item \textbf{模型}：
    \begin{itemize}
        \item 1D-CNN（卷积提取模式）
        \item LSTM/GRU（长短期记忆）
    \end{itemize}
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{block}{物理意义}
\begin{itemize}
    \item $V_{SW}$骤降 + $V_{LW}$骤升 $\Rightarrow$ 云遮挡
    \item $V_{SW}$骤升 + $V_{LW}$骤降 $\Rightarrow$ 云散开
    \item 时间序列模式识别云的移动
\end{itemize}
\end{block}

\begin{block}{应用场景}
\begin{itemize}
    \item \textbf{太阳能发电}：功率波动预警
    \item \textbf{气象预报}：短期辐射变化
    \item \textbf{舒适度监测}：体感温度预测
    \item \textbf{农业}：作物光照预测
\end{itemize}
\end{block}
\end{columns}
\end{frame}
